    74→  function req(method, url, body) {
    75→    var opts = { method: method, headers: {} };
    76→    if (body instanceof FormData) {
    77→      opts.body = body; // browser sets multipart boundary itself
    78→    } else if (body !== undefined) {
    79→      opts.headers['Content-Type'] = 'application/json';
    80→      opts.body = JSON.stringify(body);
    81→    }
    82→    return fetch(url, opts).then(function (res) {
    83→      return res.text().then(function (text) {
    84→        var data = null;
    85→        if (text) { try { data = JSON.parse(text); } catch (e) { data = null; } }
    86→        if (!res.ok) {
    87→          var detail = (data && data.detail != null) ? String(data.detail) : '';
    88→          if (res.status === 404 && /session_expired|expired|not_found|not found/i.test(detail)) {
    89→            showExpired();
    90→            var exp = new Error('session_expired');
    91→            exp.kind = 'expired'; exp.status = 404; exp.detail = detail;
    92→            throw exp;
    93→          }
    94→          var err = new Error(detail || ('http_' + res.status));
    95→          err.kind = 'http'; err.status = res.status; err.detail = detail;
    96→          throw err;
    97→        }
    98→        return data;
    99→      });
   100→    }, function () {
   101→      var net = new Error('network_error');
   102→      net.kind = 'network';
   103→      throw net;
   104→    });
   105→  }

   392→  /** GET snapshot once, replay stored events oldest-first. */
   393→  function hydrate() {
   394→    store.hydrating = true;
   395→    return api.snapshot(store.sessionId).then(function (snap) {
   396→      snap = snap || {};
   397→      if (snap.demo === true) { store.demo = true; updateDemoBadge(); }
   398→      if (snap.level === 'beginner' || snap.level === 'expert') {
   399→        store.level = snap.level;
   400→        setLevelUI();
   401→      }
   402→      var evs = Array.isArray(snap.events) ? snap.events.slice() : [];
   403→      evs.sort(function (a, b) { return (a && a.i ? a.i : 0) - (b && b.i ? b.i : 0); });
   404→      store.eventsSeen = evs.length > 0;
   405→
   406→      var approvalPayload = null;
   407→      var seenApproval = false;
   408→      var afterApproval = false;
   409→
   410→      evs.forEach(function (ev) {
   411→        if (!ev || !ev.type) return;
   412→        if (ev.type === 'approval_required') {
   413→          approvalPayload = ev.payload || {};
   414→          seenApproval = true;
   415→          afterApproval = false;
   416→        } else if (seenApproval) {
   417→          afterApproval = true;
   418→        }
   419→        handleEvent(ev.i, ev.type, ev.payload || {});
   420→      });
   421→
   422→      // Fallbacks for data the snapshot exposes outside the event list.
   423→      if (snap.plan && !store.hasPlan) handleEvent(null, 'plan', snap.plan);
   424→      if (snap.report) store.report = snap.report;
   425→
   426→      // Pending approval: snapshot field wins (note not yet shown for it);
   427→      // otherwise an approval_required that is still the newest event means
   428→      // the user has not decided yet (its note was already added by the
   429→      // event route during replay above — do not duplicate it).
   430→      var st = String(snap.status || '').toLowerCase();
   431→      if (snap.approval) {
   432→        addNote('خلبان برای ادامه نیاز به تأیید تو دارد.');
   433→        openApprovalModal(snap.approval);
   434→      } else if (approvalPayload && !afterApproval && !store.cancelled && !/done|cancel|success/.test(st)) {
   435→        openApprovalModal(approvalPayload);
   436→      }
   437→
   438→      store.hydrating = false;
   439→      scrollTimeline();
   440→      return snap;
   441→    }).catch(function (e) {
   442→      store.hydrating = false;
   443→      if (isExpired(e)) return null;
   444→      // Non-fatal: SSE replay (no Last-Event-ID yet -> full replay) covers us.
   445→      addNote('دریافت وضعیت قبلی ناموفق بود؛ ادامه می‌دهیم.');
   446→      return null;
   447→    });
   448→  }

   450→  /**
   451→   * SSE with manual reconnect: close on error, wait 2s, reopen the SAME
   452→   * URL (never add cache-busters — that would defeat Last-Event-ID replay
   453→   * semantics server-side). A freshly reopened EventSource has no
   454→   * Last-Event-ID, so the server may replay old messages; handleEvent()
   455→   * drops anything with id <= store.maxI, so nothing renders twice.
   456→   */
   457→  function openSSE() {
   458→    stopSSE();
   459→    if (!store.sessionId || store.screen === 'report' || store.screen === 'expired') return;
   460→    var es = new EventSource('/api/events/' + encodeURIComponent(store.sessionId));
   461→    store.es = es;
   462→    store.sseAttempts = 0;
   463→
   464→    es.onopen = function () { store.sseAttempts = 0; };
   465→
   466→    es.onmessage = function (ev) {
   467→      var msg = null;
   468→      try { msg = JSON.parse(ev.data); } catch (e) { return; }
   469→      if (!msg || typeof msg !== 'object' || !msg.type) return;
   470→      var payload = shallowCopy(msg);
   471→      delete payload.type;
   472→      handleEvent(ev.lastEventId, msg.type, payload);
   473→    };
   474→
   475→    es.onerror = function () {
   476→      // Native reconnect retries forever; we take over to cap attempts.
   477→      try { es.close(); } catch (e) { /* noop */ }
   478→      if (store.es === es) store.es = null;
   479→      store.sseAttempts += 1;
   480→      if (store.sseAttempts >= 10) { showExpired(); return; }
   481→      store.sseTimer = setTimeout(openSSE, 2000);
   482→    };
   483→  }
   484→
   485→  function stopSSE() {
   486→    if (store.sseTimer) { clearTimeout(store.sseTimer); store.sseTimer = null; }
   487→    if (store.es) {
   488→      try { store.es.close(); } catch (e) { /* noop */ }
