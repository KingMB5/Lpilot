    70→    70→   * API layer — every call funnels through req() so session
    71→    71→   * expiry (404 {detail:"session_expired"}) is handled once.

    73→    73→
    74→    74→  function req(method, url, body) {
    75→    75→    var opts = { method: method, headers: {} };
    76→    76→    if (body instanceof FormData) {
    77→    77→      opts.body = body; // browser sets multipart boundary itself
    78→    78→    } else if (body !== undefined) {
    79→    79→      opts.headers['Content-Type'] = 'application/json';
    80→    80→      opts.body = JSON.stringify(body);
    81→    81→    }
    82→    82→    return fetch(url, opts).then(function (res) {
    83→    83→      return res.text().then(function (text) {
    84→    84→        var data = null;
    85→    85→        if (text) { try { data = JSON.parse(text); } catch (e) { data = null; } }
    86→    86→        if (!res.ok) {
    87→    87→          var detail = (data && data.detail != null) ? String(data.detail) : '';
    88→    88→          if (res.status === 404 && /session_expired|expired|not_found|not found/i.test(detail)) {
    89→    89→            showExpired();
    90→    90→            var exp = new Error('session_expired');
    91→    91→            exp.kind = 'expired'; exp.status = 404; exp.detail = detail;
    92→    92→            throw exp;
    93→    93→          }
    94→    94→          var err = new Error(detail || ('http_' + res.status));
    95→    95→          err.kind = 'http'; err.status = res.status; err.detail = detail;
    96→    96→          throw err;
    97→    97→        }
    98→    98→        return data;
    99→    99→      });
   100→   100→    }, function () {
   101→   101→      var net = new Error('network_error');
   102→   102→      net.kind = 'network';
   103→   103→      throw net;
   104→   104→    });
   105→   105→  }

   392→   392→  /** GET snapshot once, replay stored events oldest-first. */
   393→   393→  function hydrate() {
   394→   394→    store.hydrating = true;
   395→   395→    return api.snapshot(store.sessionId).then(function (snap) {
   396→   396→      snap = snap || {};
   397→   397→      if (snap.demo === true) { store.demo = true; updateDemoBadge(); }
   398→   398→      if (snap.level === 'beginner' || snap.level === 'expert') {
   399→   399→        store.level = snap.level;
   400→   400→        setLevelUI();
   401→   401→      }
   402→   402→      var evs = Array.isArray(snap.events) ? snap.events.slice() : [];
   403→   403→      evs.sort(function (a, b) { return (a && a.i ? a.i : 0) - (b && b.i ? b.i : 0); });
   404→   404→      store.eventsSeen = evs.length > 0;
   405→   405→
   406→   406→      var approvalPayload = null;
   407→   407→      var seenApproval = false;
   408→   408→      var afterApproval = false;
   409→   409→
   410→   410→      evs.forEach(function (ev) {
   411→   411→        if (!ev || !ev.type) return;
   412→   412→        if (ev.type === 'approval_required') {
   413→   413→          approvalPayload = ev.payload || {};
   414→   414→          seenApproval = true;
   415→   415→          afterApproval = false;
   416→   416→        } else if (seenApproval) {
   417→   417→          afterApproval = true;
   418→   418→        }
   419→   419→        handleEvent(ev.i, ev.type, ev.payload || {});
   420→   420→      });
   421→   421→
   422→   422→      // Fallbacks for data the snapshot exposes outside the event list.
   423→   423→      if (snap.plan && !store.hasPlan) handleEvent(null, 'plan', snap.plan);
   424→   424→      if (snap.report) store.report = snap.report;
   425→   425→
   426→   426→      // Pending approval: snapshot field wins (note not yet shown for it);
   427→   427→      // otherwise an approval_required that is still the newest event means
   428→   428→      // the user has not decided yet (its note was already added by the
   429→   429→      // event route during replay above — do not duplicate it).
   430→   430→      var st = String(snap.status || '').toLowerCase();
   431→   431→      if (snap.approval) {
   432→   432→        addNote('خلبان برای ادامه نیاز به تأیید تو دارد.');
   433→   433→        openApprovalModal(snap.approval);
   434→   434→      } else if (approvalPayload && !afterApproval && !store.cancelled && !/done|cancel|success/.test(st)) {
   435→   435→        openApprovalModal(approvalPayload);
   436→   436→      }
   437→   437→
   438→   438→      store.hydrating = false;
   439→   439→      scrollTimeline();
   440→   440→      return snap;
   441→   441→    }).catch(function (e) {
   442→   442→      store.hydrating = false;
   443→   443→      if (isExpired(e)) return null;
   444→   444→      // Non-fatal: SSE replay (no Last-Event-ID yet -> full replay) covers us.
   445→   445→      addNote('دریافت وضعیت قبلی ناموفق بود؛ ادامه می‌دهیم.');
   446→   446→      return null;
   447→   447→    });
   448→   448→  }

   450→   450→  /**
   451→   451→   * SSE with manual reconnect: close on error, wait 2s, reopen the SAME
   452→   452→   * URL (never add cache-busters — that would defeat Last-Event-ID replay
   453→   453→   * semantics server-side). A freshly reopened EventSource has no
   454→   454→   * Last-Event-ID, so the server may replay old messages; handleEvent()
   455→   455→   * drops anything with id <= store.maxI, so nothing renders twice.
   456→   456→   */
   457→   457→  function openSSE() {
   458→   458→    stopSSE();
   459→   459→    if (!store.sessionId || store.screen === 'report' || store.screen === 'expired') return;
   460→   460→    var es = new EventSource('/api/events/' + encodeURIComponent(store.sessionId));
   461→   461→    store.es = es;
   462→   462→    store.sseAttempts = 0;
   463→   463→
   464→   464→    es.onopen = function () { store.sseAttempts = 0; };
   465→   465→
   466→   466→    es.onmessage = function (ev) {
   467→   467→      var msg = null;
   468→   468→      try { msg = JSON.parse(ev.data); } catch (e) { return; }
   469→   469→      if (!msg || typeof msg !== 'object' || !msg.type) return;
   470→   470→      var payload = shallowCopy(msg);
   471→   471→      delete payload.type;
   472→   472→      handleEvent(ev.lastEventId, msg.type, payload);
   473→   473→    };
   474→   474→
   475→   475→    es.onerror = function () {
   476→   476→      // Native reconnect retries forever; we take over to cap attempts.
   477→   477→      try { es.close(); } catch (e) { /* noop */ }
   478→   478→      if (store.es === es) store.es = null;
   479→   479→      store.sseAttempts += 1;
   480→   480→      if (store.sseAttempts >= 10) { showExpired(); return; }
   481→   481→      store.sseTimer = setTimeout(openSSE, 2000);
   482→   482→    };
   483→   483→  }
   484→   484→
   485→   485→  function stopSSE() {
   486→   486→    if (store.sseTimer) { clearTimeout(store.sseTimer); store.sseTimer = null; }
   487→   487→    if (store.es) {
   488→   488→      try { store.es.close(); } catch (e) { /* noop */ }
   489→   489→      store.es = null;
   490→   490→    }
   491→   491→  }

   493→   493→  /** Central event gate: dedupes by integer id, then routes by type. */
   494→   494→  function handleEvent(rawId, type, payload) {
   495→   495→    if (rawId !== undefined && rawId !== null && String(rawId) !== '') {
   496→   496→      var n = Number(rawId);
   497→   497→      if (Number.isFinite(n)) {
   498→   498→        if (n <= store.maxI) return; // already rendered (hydration / replay)
   499→   499→        store.maxI = n;
   500→   500→      }
   501→   501→    }
   502→   502→    var fn = ROUTES[type];
   503→   503→    if (fn) fn(payload || {});
   504→   504→  }
